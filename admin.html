Settings & Database SQL
📜 Full WooCommerce Supabase Database Schema

Copy and paste this in Supabase → SQL Editor → Run to enable all backend features (Orders, Coupons, Reviews, Settings, Products, Categories):

-- ===== CRAFTEDCORE FULL ECOMMERCE SCHEMA ===== -- -- 1. PRODUCTS create table if not exists products ( id bigint generated by default as identity primary key, name text not null, price numeric not null, category_slug text, category_name text, badge text, description text, image_url text, wa_message text, is_featured boolean default false, created_at timestamp with time zone default now() ); -- 2. CATEGORIES create table if not exists categories ( id bigint generated by default as identity primary key, name text not null, slug text unique not null, emoji text, sort_order int default 0, created_at timestamp with time zone default now() ); -- 3. ORDERS create table if not exists orders ( id bigint generated by default as identity primary key, customer_name text not null, customer_phone text, product_name text, amount numeric default 0, status text default 'pending', notes text, created_at timestamp with time zone default now() ); -- 4. REVIEWS create table if not exists reviews ( id bigint generated by default as identity primary key, author_name text not null, location text, rating int default 5, review_text text, is_active boolean default true, created_at timestamp with time zone default now() ); -- 5. COUPONS create table if not exists coupons ( id bigint generated by default as identity primary key, code text unique not null, discount_percent int default 10, is_active boolean default true, created_at timestamp with time zone default now() ); -- 6. SITE SETTINGS create table if not exists site_settings ( key text primary key, value text, updated_at timestamp with time zone default now() ); -- RLS POLICIES (ENABLE PUBLIC READ/WRITE) alter table products enable row level security; alter table categories enable row level security; alter table orders enable row level security; alter table reviews enable row level security; alter table coupons enable row level security; alter table site_settings enable row level security; create policy "Public Products" on products for all using (true) with check (true); create policy "Public Categories" on categories for all using (true) with check (true); create policy "Public Orders" on orders for all using (true) with check (true); create policy "Public Reviews" on reviews for all using (true) with check (true); create policy "Public Coupons" on coupons for all using (true) with check (true); create policy "Public Settings" on site_settings for all using (true) with check (true); -- 7. VISITOR ANALYTICS & TRAFFIC create table if not exists analytics ( id bigint generated always as identity primary key, visitor_id text not null, session_id text, page_path text, page_name text, referrer text, traffic_source text, device_type text, browser text, os text, screen_width int, screen_height int, country text default 'India', city text, region text, timestamp timestamp with time zone default now() ); -- 8. LIVE ACTIVE VISITORS HEARTBEAT create table if not exists analytics_live ( visitor_id text primary key, session_id text, page_path text, page_name text, device_type text, traffic_source text, country text default 'India', city text, last_seen timestamp with time zone default now() ); alter table analytics enable row level security; alter table analytics_live enable row level security; create policy "Public Analytics" on analytics for all using (true) with check (true); create policy "Public Analytics Live" on analytics_live for all using (true) with check (true);
