// ─── QUICK ORDERS ────────────────────────────────────────────────
let quickOrders = [];

async function loadQO() {
  const settings = await SiteSettings.getAll();
  quickOrders = settings.quick_orders ? JSON.parse(settings.quick_orders) : [];
  renderQO();
}

function renderQO() {
  if (quickOrders.length === 0) {
    document.getElementById('qoTbl').innerHTML = '<div class="empty">No Quick Order items found.</div>';
    return;
  }
  let h = '<table><thead><tr><th>Icon</th><th>Title</th><th>Description</th><th>Price</th><th>Message</th><th>Actions</th></tr></thead><tbody>';
  quickOrders.forEach((q, i) => {
    const iconHtml = (q.icon && (q.icon.startsWith('http') || q.icon.includes('/'))) 
      ? `<img src="${q.icon}" style="width:40px;height:40px;object-fit:cover;border-radius:6px;" />` 
      : (q.icon || '');
    h += `<tr>
      <td style="font-size:1.5rem; text-align:center;">${iconHtml}</td>
      <td style="font-weight:600;">${q.title || ''}</td>
      <td style="font-size:0.8rem; color:var(--text2); max-width:180px;">${q.desc || '<span style="color:var(--text3);font-style:italic;">—</span>'}</td>
      <td>${q.price || ''}</td>
      <td style="font-size:0.8rem; color:var(--text2);">${q.msg || ''}</td>
      <td>
        <div class="act-cell">
          <button class="btn btn-icon" style="background:rgba(200,138,88,0.15); color:var(--gold);" onclick="openQOModal(${i})">✏️</button>
          <button class="btn btn-icon" style="background:rgba(224,85,85,0.15); color:var(--red);" onclick="delQO(${i})">🗑️</button>
        </div>
      </td>
    </tr>`;
  });
  h += '</tbody></table>';
  document.getElementById('qoTbl').innerHTML = h;
}

function openQOModal(i = -1) {
  if (i === -1 || i === '-1') {
    document.getElementById('qoModalTitle').innerText = 'Add Quick Order Item';
    document.getElementById('qoId').value = '';
    document.getElementById('qoTitle').value = '';
    document.getElementById('qoDesc').value = '';
    document.getElementById('qoPrice').value = '';
    document.getElementById('qoIcon').value = '';
    document.getElementById('qoMsg').value = '';
  } else {
    const q = quickOrders[i];
    if (q) {
      document.getElementById('qoModalTitle').innerText = 'Edit Quick Order Item';
      document.getElementById('qoId').value = i;
      document.getElementById('qoTitle').value = q.title || '';
      document.getElementById('qoDesc').value = q.desc || '';
      document.getElementById('qoPrice').value = q.price || '';
      document.getElementById('qoIcon').value = q.icon || '';
      document.getElementById('qoMsg').value = q.msg || '';
    }
  }
  openM('qoModal');
}

async function saveQO() {
  const i = document.getElementById('qoId').value;
  const q = {
    title: document.getElementById('qoTitle').value.trim(),
    desc: document.getElementById('qoDesc').value.trim(),
    price: document.getElementById('qoPrice').value.trim(),
    icon: document.getElementById('qoIcon').value.trim(),
    msg: document.getElementById('qoMsg').value.trim()
  };
  
  if (!q.title) return toast('Please enter a title', 'error');
  
  const btn = document.getElementById('qoSaveBtn');
  if (btn) {
    btn.disabled = true;
    btn.innerText = 'Saving...';
  }
  
  if (i === '') quickOrders.push(q);
  else quickOrders[i] = q;
  
  try {
    await SiteSettings.set('quick_orders', JSON.stringify(quickOrders));
    closeM('qoModal');
    toast('Item saved successfully!', 'success');
    renderQO();
  } catch(e) {
    console.error(e);
    toast('❌ DB Error: ' + e.message, 'error');
  } finally {
    if (btn) {
      btn.disabled = false;
      btn.innerText = '💾 Save Item';
    }
  }
}

function delQO(i) {
  if(confirm('Are you sure you want to delete this item?')) {
    quickOrders.splice(i, 1);
    SiteSettings.set('quick_orders', JSON.stringify(quickOrders)).then(() => {
      renderQO();
      toast('Item deleted!', 'success');
    }).catch(e => {
      console.error(e);
      toast('❌ DB Error: ' + e.message, 'error');
    });
  }
}

// ─── INIT ────────────────────────────────────────────────────────
document.addEventListener('DOMContentLoaded', () => {
  loadQO();
});
