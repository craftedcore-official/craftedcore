  <script>
    document.addEventListener('DOMContentLoaded', async () => {
      try {
        const settings = await SiteSettings.getAll();
        const grid = document.getElementById('quickOrderGrid');
        if (settings.quick_orders) {
          const qo = JSON.parse(settings.quick_orders);
          if (qo.length > 0) {
            grid.classList.toggle('has-single', qo.length === 1);
            let html = '';
            qo.forEach(q => {
              const hasImg = q.icon && (q.icon.startsWith('http') || q.icon.includes('/') || q.icon.startsWith('data:'));
              const iconHtml = hasImg 
                ? `<img src="${q.icon}" alt="${escapeHtml(q.title || 'Product')}" class="qo-img" loading="lazy" />` 
                : `<div class="qo-emoji">${q.icon || '👕'}</div>`;
              
              const rawPrice = (q.price || '').trim();
              let displayPrice = rawPrice;
              if (rawPrice && !rawPrice.startsWith('₹') && !rawPrice.toLowerCase().includes('from') && !rawPrice.toLowerCase().includes('rs')) {
                displayPrice = '₹' + rawPrice;
              }
              
              const msg = q.msg || ('Hi CraftedCore! I want to order ' + (q.title || 'this item') + '.');
              const waLink = `https://wa.me/+918320979383?text=${encodeURIComponent(msg)}`;
              
              html += `
              <a href="${waLink}" target="_blank" class="qo-card" title="Order ${escapeHtml(q.title || '')} on WhatsApp">
                <div class="qo-img-wrap">
                  <span class="qo-badge">⚡ Quick Order</span>
                  ${iconHtml}
                </div>
                <div class="qo-body">
                  <div class="qo-title">${escapeHtml(q.title || 'Custom Product')}</div>
                  <div class="qo-bottom">
                    <div class="qo-price">${escapeHtml(displayPrice)}</div>
                    <span class="qo-btn">
                      <svg viewBox="0 0 24 24" width="14" height="14" fill="currentColor"><path d="M12.012 2c-5.506 0-9.989 4.478-9.989 9.984 0 1.758.459 3.474 1.33 4.982L2 22l5.176-1.348a9.982 9.982 0 004.836 1.251h.004c5.506 0 9.989-4.478 9.989-9.984 0-2.668-1.039-5.176-2.927-7.063C17.189 3.039 14.68 2 12.012 2zm.004 16.326h-.003a8.31 8.31 0 01-4.238-1.163l-.304-.18-3.149.821.838-3.067-.198-.314a8.275 8.275 0 01-1.272-4.437c0-4.577 3.724-8.301 8.306-8.301 2.217 0 4.301.865 5.869 2.434a8.261 8.261 0 012.43 5.867c0 4.578-3.725 8.302-8.279 8.302zm4.551-6.216c-.249-.125-1.477-.729-1.706-.812-.228-.083-.395-.125-.561.125-.166.249-.644.812-.789.978-.145.166-.291.187-.54.062a6.837 6.837 0 01-2.008-1.238 7.55 7.55 0 01-1.39-1.73c-.145-.249-.016-.384.109-.508.113-.112.249-.291.374-.436.125-.145.166-.249.249-.415.083-.166.042-.312-.021-.436-.062-.125-.561-1.352-.769-1.85-.202-.486-.407-.42-.561-.428l-.478-.009c-.166 0-.436.062-.664.312-.228.249-.873.852-.873 2.079 0 1.226.893 2.41 1.018 2.577.125.166 1.757 2.684 4.257 3.764.595.257 1.06.41 1.422.525.597.19 1.14.163 1.57.099.479-.071 1.477-.603 1.684-1.184.208-.582.208-1.08.145-1.184-.063-.104-.229-.166-.478-.291z"/></svg>
                      Order Now
                    </span>
                  </div>
                </div>
              </a>`;
            });
            grid.innerHTML = html;
          } else {
            grid.innerHTML = '<div style="text-align:center; padding:2rem; width:100%; color:var(--text2);">No products found. Add some from the Admin Panel.</div>';
          }
        } else {
          grid.innerHTML = '<div style="text-align:center; padding:2rem; width:100%; color:var(--text2);">No products found. Add some from the Admin Panel.</div>';
        }
      } catch(e) { 
        console.error(e); 
        document.getElementById('quickOrderGrid').innerHTML = '<div style="text-align:center; padding:2rem; width:100%; color:var(--red);">Failed to load products.</div>';
      }
    });
  </script>
</body>
</html>
